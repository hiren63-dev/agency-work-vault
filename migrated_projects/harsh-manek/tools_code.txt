// Income Tax Estimator
function IncomeTaxCalculator() {
  const [income, setIncome] = useState(1200000);
  const [regime, setRegime] = useState<"new" | "old">("new");

  const calcNew = (inc: number) => {
    if (inc <= 300000) return 0;
    if (inc <= 700000) return (inc - 300000) * 0.05;
    if (inc <= 1000000) return 20000 + (inc - 700000) * 0.10;
    if (inc <= 1200000) return 50000 + (inc - 1000000) * 0.15;
    if (inc <= 1500000) return 80000 + (inc - 1200000) * 0.20;
    return 110000 + (inc - 1500000) * 0.30;
  };

  const calcOld = (inc: number) => {
    const taxable = Math.max(0, inc - 150000); // basic 80C
    if (taxable <= 250000) return 0;
    if (taxable <= 500000) return (taxable - 250000) * 0.05;
    if (taxable <= 1000000) return 12500 + (taxable - 500000) * 0.20;
    return 112500 + (taxable - 1000000) * 0.30;
  };

  const tax = regime === "new" ? calcNew(income) : calcOld(income);
  const cess = tax * 0.04;
  const total = tax + cess;
  const effectiveRate = income > 0 ? ((total / income) * 100).toFixed(1) : "0.0";

  return (
    <div className="glass-card p-6 sm:p-8">
      <div className="flex items-center gap-3 mb-6">
        <div className="w-9 h-9 rounded-full bg-white/10 flex items-center justify-center border border-white/10">
          <Calculator className="w-4 h-4 text-white/70" />
        </div>
        <div>
          <h3 className="font-sans font-semibold text-white text-base">Income Tax Estimator</h3>
          <p className="text-[11px] font-mono text-white/40 uppercase tracking-wider">FY 2025-26 · AY 2026-27</p>
        </div>
      </div>

      {/* Regime toggle */}
      <div className="flex gap-2 mb-5 p-1 rounded-full bg-white/5 border border-white/10 w-fit">
        {(["new", "old"] as const).map((r) => (
          <button
            key={r}
            onClick={() => setRegime(r)}
            className={`relative px-5 py-2 rounded-full text-sm font-sans font-medium transition-colors capitalize ${regime === r ? "text-black" : "text-white/50 hover:text-white"}`}
          >
            {regime === r && <motion.div layoutId="regimeTab" className="absolute inset-0 bg-white rounded-full" transition={SPRING_MICRO} />}
            <span className="relative z-10">{r} Regime</span>
          </button>
        ))}
      </div>

      <label className="block font-mono text-[11px] uppercase tracking-widest text-white/40 mb-2">Annual Income</label>
      <input
        type="range"
        min={0}
        max={5000000}
        step={50000}
        value={income}
        onChange={(e) => setIncome(Number(e.target.value))}
        className="w-full accent-white mb-2"
      />
      <div className="flex justify-between text-xs font-mono text-white/40 mb-6">
        <span>₹0</span>
        <span className="text-white font-semibold text-sm">₹{income.toLocaleString("en-IN")}</span>
        <span>₹50L</span>
      </div>

      <div className="grid grid-cols-3 gap-3 mb-5">
        {[
          { label: "Base Tax", value: `₹${Math.round(tax).toLocaleString("en-IN")}` },
          { label: "Health & Edu Cess (4%)", value: `₹${Math.round(cess).toLocaleString("en-IN")}` },
          { label: "Total Payable", value: `₹${Math.round(total).toLocaleString("en-IN")}` },
        ].map(({ label, value }) => (
          <div key={label} className="p-3 rounded-xl bg-white/[0.04] border border-white/[0.08] text-center">
            <span className="block font-sans font-bold text-white text-sm">{value}</span>
            <span className="block font-mono text-[9px] text-white/35 uppercase tracking-wider mt-0.5">{label}</span>
          </div>
        ))}
      </div>

      <div className="flex items-center justify-between p-4 rounded-xl bg-white/[0.06] border border-white/10">
        <span className="font-mono text-xs text-white/50 uppercase tracking-widest">Effective Rate</span>
        <span className="font-sans font-bold text-white text-xl">{effectiveRate}%</span>
      </div>

      <a
        href={WA_BASE}
        target="_blank"
        rel="noopener noreferrer"
        className="mt-4 flex items-center justify-center gap-2 w-full h-11 rounded-xl border border-white/10 text-white/60 hover:text-white hover:border-white/25 font-sans text-sm transition-colors"
      >
        Get a personalised tax plan <ArrowRight className="w-4 h-4" />
      </a>
    </div>
  );
}

// GST Calculator
function GSTCalculator() {
  const [amount, setAmount] = useState("");
  const [rate, setRate] = useState(18);
  const [mode, setMode] = useState<"exclusive" | "inclusive">("exclusive");

  const numAmount = parseFloat(amount) || 0;
  const gstRates = [0, 5, 12, 18, 28];

  let base = 0, gst = 0, total = 0;
  if (mode === "exclusive") {
    base = numAmount;
    gst = (numAmount * rate) / 100;
    total = base + gst;
  } else {
    total = numAmount;
    base = (numAmount * 100) / (100 + rate);
    gst = total - base;
  }

  return (
    <div className="glass-card p-6 sm:p-8">
      <div className="flex items-center gap-3 mb-6">
        <div className="w-9 h-9 rounded-full bg-white/10 flex items-center justify-center border border-white/10">
          <PieChart className="w-4 h-4 text-white/70" />
        </div>
        <div>
          <h3 className="font-sans font-semibold text-white text-base">GST Calculator</h3>
          <p className="text-[11px] font-mono text-white/40 uppercase tracking-wider">CGST + SGST / IGST breakdown</p>
        </div>
      </div>

      <div className="flex gap-2 mb-5 p-1 rounded-full bg-white/5 border border-white/10 w-fit">
        {(["exclusive", "inclusive"] as const).map((m) => (
          <button
            key={m}
            onClick={() => setMode(m)}
            className={`relative px-4 py-2 rounded-full text-xs font-mono uppercase tracking-wider transition-colors ${mode === m ? "text-black" : "text-white/50 hover:text-white"}`}
          >
            {mode === m && <motion.div layoutId="gstTab" className="absolute inset-0 bg-white rounded-full" transition={SPRING_MICRO} />}
            <span className="relative z-10">GST {m}</span>
          </button>
        ))}
      </div>

      <label className="block font-mono text-[11px] uppercase tracking-widest text-white/40 mb-2">Amount (₹)</label>
      <input
        type="number"
        value={amount}
        onChange={(e) => setAmount(e.target.value)}
        placeholder="Enter amount"
        className="w-full bg-black/40 border border-white/10 rounded-xl px-4 py-3 text-sm font-sans text-white placeholder-white/30 focus:outline-none focus:border-white/30 transition-colors mb-4"
      />

      <label className="block font-mono text-[11px] uppercase tracking-widest text-white/40 mb-2">GST Rate</label>
      <div className="flex gap-2 flex-wrap mb-6">
        {gstRates.map((r) => (
          <button
            key={r}
            onClick={() => setRate(r)}
            className={`px-3.5 py-1.5 rounded-full text-xs font-mono font-semibold transition-colors ${rate === r ? "bg-white text-black" : "bg-white/[0.06] text-white/50 border border-white/10 hover:text-white"}`}
          >
            {r}%
          </button>
        ))}
      </div>

      {numAmount > 0 && (
        <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={SPRING_SNAPPY} className="grid grid-cols-2 gap-3">
          {[
            { label: "Base Amount", value: `₹${base.toFixed(2)}` },
            { label: `GST @${rate}%`, value: `₹${gst.toFixed(2)}` },
            { label: "CGST", value: `₹${(gst / 2).toFixed(2)}` },
            { label: "SGST / IGST", value: `₹${(gst / 2).toFixed(2)}` },
          ].map(({ label, value }) => (
            <div key={label} className="p-3 rounded-xl bg-white/[0.04] border border-white/[0.08]">
              <span className="block font-sans font-bold text-white text-sm">{value}</span>
              <span className="block font-mono text-[9px] text-white/35 uppercase tracking-wider mt-0.5">{label}</span>
            </div>
          ))}
          <div className="col-span-2 p-4 rounded-xl bg-white/[0.08] border border-white/15 flex items-center justify-between">
            <span className="font-mono text-xs text-white/60 uppercase tracking-widest">Total Payable</span>
            <span className="font-sans font-bold text-white text-xl">₹{total.toFixed(2)}</span>
          </div>
        </motion.div>
      )}
    </div>
  );
}
